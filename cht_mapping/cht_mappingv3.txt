import os
import re
import csv
from pathlib import Path

def detect_delimiter_and_encoding(file_path):
    """
    Detect encoding and delimiter (comma, semicolon, tab, pipe) of a CSV file.
    """
    encodings = ["utf-8-sig", "utf-8", "cp1252", "latin-1", "utf-16"]
    
    for enc in encodings:
        try:
            with open(file_path, "r", encoding=enc, errors="strict") as f:
                sample = f.read(8192)
                if not sample:
                    return enc, ","
                
                first_line = sample.splitlines()[0] if sample.splitlines() else ""
                
                try:
                    dialect = csv.Sniffer().sniff(sample, delimiters=[",", ";", "\t", "|"])
                    return enc, dialect.delimiter
                except Exception:
                    counts = {d: first_line.count(d) for d in [",", ";", "\t", "|"]}
                    best_delim = max(counts, key=counts.get)
                    return enc, best_delim if counts[best_delim] > 0 else ","
        except (UnicodeDecodeError, Exception):
            continue
            
    return "utf-8-sig", ","

def get_col_value(row, possible_names, default=""):
    """
    Find value from a CSV row dictionary matching any of the possible column names.
    Order of matching:
    1. Exact match (case-insensitive & stripped)
    2. Substring match (only for specific long pattern names)
    """
    if not row:
        return default

    # Clean keys dictionary: { 'cleaned_lower_key': 'original_key' }
    row_keys_cleaned = {}
    for k in row.keys():
        if k is not None:
            clean_k = str(k).strip().strip('\ufeff').strip('"').strip("'").lower()
            row_keys_cleaned[clean_k] = k
    
    # 1. Try exact matches first for all possible names
    for name in possible_names:
        target = name.strip().lower()
        if target in row_keys_cleaned:
            actual_key = row_keys_cleaned[target]
            val = row[actual_key]
            return str(val).strip() if val is not None else ""

    # 2. Try substring match (only if target name is at least 4 characters to avoid false matches)
    for target_name in possible_names:
        target = target_name.strip().lower()
        if len(target) < 4:
            continue
        for key_clean, actual_key in row_keys_cleaned.items():
            if target in key_clean:
                val = row[actual_key]
                return str(val).strip() if val is not None else ""

    return default

def main():
    script_dir = Path(__file__).parent.resolve()
    policy_file = script_dir / "policy.csv"
    change_file = script_dir / "change.csv"
    result_file = script_dir / "result.csv"

    print("=" * 70)
    print("CHT Mapping Tool")
    print(f"Working Directory: {script_dir}")
    print("=" * 70)

    if not policy_file.exists():
        print(f"\n[ERROR] '{policy_file.name}' not found in {script_dir}")
        print("Please make sure policy.csv is located in the same directory as this script.")
        return

    if not change_file.exists():
        print(f"\n[ERROR] '{change_file.name}' not found in {script_dir}")
        print("Please make sure change.csv is located in the same directory as this script.")
        return

    # ----------------------------------------------------
    # Step 1: Load change.csv
    # ----------------------------------------------------
    enc_change, delim_change = detect_delimiter_and_encoding(change_file)
    print(f"\n[INFO] Reading {change_file.name}")
    print(f"       Encoding: {enc_change} | Delimiter: {repr(delim_change)}")
    
    change_lookup = {}
    with open(change_file, mode="r", encoding=enc_change, errors="replace") as f:
        reader = csv.DictReader(f, delimiter=delim_change)
        headers = [h.strip() for h in (reader.fieldnames or []) if h]
        print(f"       Headers in change.csv: {headers}")

        for row in reader:
            cht_no = get_col_value(row, ["change number", "change_number", "cht", "cht number", "change ticket", "number", "ticket"])
            if cht_no:
                cht_match = re.search(r'CHT-?\d{7,8}', cht_no, re.IGNORECASE)
                cht_key = cht_match.group(0).upper().replace("-", "") if cht_match else cht_no.upper().replace("-", "")
                
                requester = get_col_value(row, ["requester", "requster", "requested_by", "requested by", "opened_by", "opened by", "author"])
                requester_group = get_col_value(row, ["requester_group", "requester group", "requster_group", "request_group", "assignment group", "group"])
                short_desc = get_col_value(row, ["short description", "short_description", "description", "summary"])
                
                change_lookup[cht_key] = {
                    "requester": requester,
                    "requester_group": requester_group,
                    "short_description": short_desc
                }

    print(f"       Loaded {len(change_lookup)} change records.")

    # ----------------------------------------------------
    # Step 2: Process policy.csv
    # ----------------------------------------------------
    enc_policy, delim_policy = detect_delimiter_and_encoding(policy_file)
    print(f"\n[INFO] Reading {policy_file.name}")
    print(f"       Encoding: {enc_policy} | Delimiter: {repr(delim_policy)}")

    # CHT Regex (CHT followed by 7 or 8 digits, optional hyphen)
    cht_pattern = re.compile(r'\bCHT-?\d{7,8}\b', re.IGNORECASE)

    # Alias definitions for SecureTrack / Policy CSV columns
    policy_name_aliases = ["policy name", "policy_name", "policy", "security policy", "device name", "device", "package", "binding", "firewall", "management"]
    rule_id_aliases = ["securetrack rule id", "rule id", "rule_id", "securetrack_rule_id", "rule number", "rule_number", "rule #", "rule#", "id", "#", "no.", "number"]
    rule_name_aliases = ["rule name", "rule_name", "rule label", "rule_label", "name", "title", "rule title"]
    comment_aliases = ["comment", "comments", "description", "note", "notes", "remark", "remarks", "documentation", "cht"]

    output_rows = []
    total_policy_rows = 0
    extracted_cht_count = 0

    with open(policy_file, mode="r", encoding=enc_policy, errors="replace") as f:
        reader = csv.DictReader(f, delimiter=delim_policy)
        raw_headers = reader.fieldnames or []
        headers = [str(h).strip() for h in raw_headers if h]
        print(f"       Headers in policy.csv: {headers}")

        # Show detected column matches for diagnostics
        sample_row = next(reader, None)
        if sample_row:
            print("\n--- Diagnostic Column Matching for First Row in policy.csv ---")
            print(f"  Detected Policy Name -> '{get_col_value(sample_row, policy_name_aliases)}'")
            print(f"  Detected Rule ID    -> '{get_col_value(sample_row, rule_id_aliases)}'")
            print(f"  Detected Rule Name  -> '{get_col_value(sample_row, rule_name_aliases)}'")
            print(f"  Detected Comment    -> '{get_col_value(sample_row, comment_aliases)}'")
            print("----------------------------------------------------------------\n")
            
            # Reset file reader position back to top (after header)
            f.seek(0)
            reader = csv.DictReader(f, delimiter=delim_policy)

        for row in reader:
            total_policy_rows += 1
            
            policy_name = get_col_value(row, policy_name_aliases)
            rule_id = get_col_value(row, rule_id_aliases)
            rule_name = get_col_value(row, rule_name_aliases)
            comment = get_col_value(row, comment_aliases)

            # Fallback text search across row if comment column was empty or unparsed
            search_text = comment if comment else " ".join(str(v) for v in row.values() if v)

            # Extract all CHT numbers
            cht_matches = cht_pattern.findall(search_text)
            
            if cht_matches:
                seen_chts = set()
                for cht in cht_matches:
                    cht_formatted = cht.upper().replace("-", "")
                    if cht_formatted not in seen_chts:
                        seen_chts.add(cht_formatted)
                        extracted_cht_count += 1

                        change_info = change_lookup.get(cht_formatted, {})
                        requester = change_info.get("requester", "")
                        requester_group = change_info.get("requester_group", "")

                        output_rows.append({
                            "Policy Name": policy_name,
                            "SecureTrack Rule ID": rule_id,
                            "Rule Name": rule_name,
                            "Change Number": cht_formatted,
                            "Requester": requester,
                            "Requester Group": requester_group
                        })

    # ----------------------------------------------------
    # Step 3: Write result.csv
    # ----------------------------------------------------
    fieldnames = [
        "Policy Name",
        "SecureTrack Rule ID",
        "Rule Name",
        "Change Number",
        "Requester",
        "Requester Group"
    ]

    with open(result_file, mode="w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(output_rows)

    print("=" * 70)
    print("SUMMARY RESULTS")
    print("=" * 70)
    print(f"Total policy rows processed: {total_policy_rows}")
    print(f"Total CHT numbers extracted: {extracted_cht_count}")
    print(f"Output saved to: {result_file.name} ({len(output_rows)} rows)")
    print("=" * 70)

if __name__ == "__main__":
    main()
